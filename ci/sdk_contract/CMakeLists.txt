cmake_minimum_required(VERSION 3.25)
project(TurboP2PSdkContract LANGUAGES C CXX)
include(CTest)

# No source-tree injection, retired package aliases, or independent DataBind package.
foreach(_entry IN ITEMS "Salts|SALTS_ROOT" "SaltsUtils|SALTS_UTILS_ROOT" "Chttp|CHTTP_ROOT")
  string(REPLACE "|" ";" _fields "${_entry}")
  list(GET _fields 0 _package)
  list(GET _fields 1 _variable)
  if("$ENV{${_variable}}" STREQUAL "")
    message(FATAL_ERROR "${_variable} must identify the restored installed SDK")
  endif()
  file(TO_CMAKE_PATH "$ENV{${_variable}}" _root)
  set(_config_dir "${_root}/lib/cmake/${_package}")
  if(NOT EXISTS "${_config_dir}/${_package}Config.cmake")
    message(FATAL_ERROR "Missing ${_package}Config.cmake under ${_root}")
  endif()
  set(${_package}_DIR "${_config_dir}" CACHE PATH "Explicit restored SDK" FORCE)
  find_package(${_package} CONFIG REQUIRED PATHS "${_config_dir}"
               NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
endforeach()

set(_required_targets Salts::CNet Salts::CMeta Salts::CFlow Salts::PluginABI
    Salts::Plugin Salts::DataBind CHttp::Client CHttp::Server)
foreach(_target IN LISTS _required_targets)
  if(NOT TARGET ${_target})
    message(FATAL_ERROR "Latest restored SDK graph is missing ${_target}")
  endif()
endforeach()
foreach(_target IN ITEMS CHttp::Client CHttp::Server)
  get_target_property(_links ${_target} INTERFACE_LINK_LIBRARIES)
  if(NOT "Salts::CNet" IN_LIST _links)
    message(FATAL_ERROR "${_target} must expose its canonical Salts::CNet dependency")
  endif()
endforeach()

option(TURBOP2P_CONTRACT_SANITIZERS "Instrument consumer tests, not prebuilt SDKs" OFF)
foreach(_name IN ITEMS headers cnet_smoke)
  add_executable(turbop2p_sdk_${_name} ${_name}.c)
  target_compile_features(turbop2p_sdk_${_name} PRIVATE c_std_11)
  target_link_libraries(turbop2p_sdk_${_name} PRIVATE ${_required_targets})
  if(MSVC)
    target_compile_options(turbop2p_sdk_${_name} PRIVATE /experimental:c11atomics /W4)
  else()
    target_compile_options(turbop2p_sdk_${_name} PRIVATE -Wall -Wextra -Werror)
  endif()
  if(TURBOP2P_CONTRACT_SANITIZERS)
    if(MSVC OR NOT CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
      message(FATAL_ERROR "This contract sanitizer preset requires Unix Clang/GCC")
    endif()
    target_compile_options(turbop2p_sdk_${_name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(turbop2p_sdk_${_name} PRIVATE -fsanitize=address,undefined)
  endif()
  if(WIN32)
    # Client/Server are shared imports; use their configuration-specific DLL closure.
    add_custom_command(TARGET turbop2p_sdk_${_name} POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different
        $<TARGET_RUNTIME_DLLS:turbop2p_sdk_${_name}>
        $<TARGET_FILE_DIR:turbop2p_sdk_${_name}>
      COMMAND_EXPAND_LISTS VERBATIM)
  endif()
  add_test(NAME sdk_${_name} COMMAND turbop2p_sdk_${_name})
  set_tests_properties(sdk_${_name} PROPERTIES TIMEOUT 30)
endforeach()
